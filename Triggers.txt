SQL> CREATE OR REPLACE TRIGGER trg_failed_login_alert
  2  FOR INSERT ON login_audit
  3  COMPOUND TRIGGER
  4
  5      -- Variables that persist for the statement
  6      TYPE user_list_t IS TABLE OF VARCHAR2(100);
  7      failed_users user_list_t := user_list_t();
  8
  9  BEFORE EACH ROW IS
 10  BEGIN
 11      -- Only track failed attempts
 12      IF :NEW.status = 'FAILED' THEN
 13          failed_users.EXTEND;
 14          failed_users(failed_users.COUNT) := :NEW.username;
 15      END IF;
 16  END BEFORE EACH ROW;
 17
 18  AFTER STATEMENT IS
 19      failed_count NUMBER;
 20  BEGIN
 21      -- Loop through all users who had failed attempts in this statement
 22      FOR i IN 1 .. failed_users.COUNT LOOP
 23
 24          -- Count failed attempts for the same user today
 25          SELECT COUNT(*)
 26          INTO failed_count
 27          FROM login_audit
 28          WHERE username = failed_users(i)
 29            AND status = 'FAILED'
 30            AND TRUNC(attempt_time) = TRUNC(SYSDATE);
 31
 32          -- If failed attempts exceed 2, insert alert
 33          IF failed_count > 2 THEN
 34              INSERT INTO security_alerts (
 35                  username,
 36                  failed_attempts,
 37                  alert_time,
 38                  alert_message,
 39                  contact_email
 40              ) VALUES (
 41                  failed_users(i),
 42                  failed_count,
 43                  SYSDATE,
 44                  'Suspicious login activity: ' || failed_count || ' failed attempts.',
 45                  'security-team@company.com'
 46              );
 47          END IF;
 48
 49      END LOOP;
 50  END AFTER STATEMENT;
 51
 52  END trg_failed_login_alert;
